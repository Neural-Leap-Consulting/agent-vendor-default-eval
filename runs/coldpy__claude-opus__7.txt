## `config.py`

```python
import os


def _bool(name, default="false"):
    return os.environ.get(name, default).strip().lower() in ("1", "true", "yes", "on")


class Config:
    SECRET_KEY = os.environ["SECRET_KEY"]

    SQLALCHEMY_DATABASE_URI = os.environ.get("DATABASE_URL", "sqlite:///app.db")
    SQLALCHEMY_TRACK_MODIFICATIONS = False

    APP_NAME = os.environ.get("APP_NAME", "Acme")
    APP_URL = os.environ.get("APP_URL", "http://localhost:5000")

    MAIL_SERVER = os.environ.get("MAIL_SERVER", "localhost")
    MAIL_PORT = int(os.environ.get("MAIL_PORT", 587))
    MAIL_USE_TLS = _bool("MAIL_USE_TLS", "true")    # STARTTLS, port 587
    MAIL_USE_SSL = _bool("MAIL_USE_SSL", "false")   # implicit TLS, port 465
    MAIL_USERNAME = os.environ.get("MAIL_USERNAME")
    MAIL_PASSWORD = os.environ.get("MAIL_PASSWORD")
    MAIL_DEFAULT_SENDER = os.environ.get("MAIL_DEFAULT_SENDER", "no-reply@example.com")
    MAIL_TIMEOUT = int(os.environ.get("MAIL_TIMEOUT", 10))

    # Set true in tests/dev to log emails instead of sending them.
    MAIL_SUPPRESS_SEND = _bool("MAIL_SUPPRESS_SEND", "false")
```

## `models.py`

```python
from datetime import datetime, timezone

from werkzeug.security import check_password_hash, generate_password_hash

from extensions import db


def utcnow():
    return datetime.now(timezone.utc)


class User(db.Model):
    __tablename__ = "users"

    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(255), unique=True, nullable=False, index=True)
    name = db.Column(db.String(120))
    password_hash = db.Column(db.String(255), nullable=False)
    created_at = db.Column(db.DateTime(timezone=True), default=utcnow, nullable=False)
    welcome_email_sent_at = db.Column(db.DateTime(timezone=True))

    def set_password(self, password):
        self.password_hash = generate_password_hash(password)

    def check_password(self, password):
        return check_password_hash(self.password_hash, password)
```

## `mailer.py`

```python
import logging
import smtplib
import ssl
from email.message import EmailMessage
from threading import Thread

from flask import current_app, render_template

log = logging.getLogger(__name__)


def _smtp_settings(app):
    c = app.config
    return {
        "server": c["MAIL_SERVER"],
        "port": c["MAIL_PORT"],
        "use_tls": c["MAIL_USE_TLS"],
        "use_ssl": c["MAIL_USE_SSL"],
        "username": c["MAIL_USERNAME"],
        "password": c["MAIL_PASSWORD"],
        "timeout": c["MAIL_TIMEOUT"],
        "suppress": c["MAIL_SUPPRESS_SEND"],
    }


def _deliver(settings, message):
    """Blocking SMTP delivery. Never raises — logs instead."""
    if settings["suppress"]:
        log.info("MAIL_SUPPRESS_SEND on; not sending to %s\n%s",
                 message["To"], message.get_content())
        return

    try:
        context = ssl.create_default_context()
        if settings["use_ssl"]:
            client = smtplib.SMTP_SSL(settings["server"], settings["port"],
                                      timeout=settings["timeout"], context=context)
        else:
            client = smtplib.SMTP(settings["server"], settings["port"],
                                  timeout=settings["timeout"])

        with client:
            client.ehlo()
            if settings["use_tls"] and not settings["use_ssl"]:
                client.starttls(context=context)
                client.ehlo()
            if settings["username"]:
                client.login(settings["username"], settings["password"])
            client.send_message(message)

        log.info("Sent %r to %s", message["Subject"], message["To"])
    except Exception:
        log.exception("Failed to send %r to %s", message["Subject"], message["To"])


def send_email(subject, recipient, text_body, html_body=None, sender=None, background=True):
    """Build and send an email. Rendering must happen before this is called
    (templates need an app context; the sender thread has none)."""
    app = current_app._get_current_object()

    message = EmailMessage()
    message["Subject"] = subject
    message["From"] = sender or app.config["MAIL_DEFAULT_SENDER"]
    message["To"] = recipient
    message.set_content(text_body)
    if html_body:
        message.add_alternative(html_body, subtype="html")

    settings = _smtp_settings(app)

    if background:
        Thread(target=_deliver, args=(settings, message), daemon=True).start()
    else:
        _deliver(settings, message)


def send_welcome_email(user, background=True):
    ctx = {
        "user": user,
        "app_name": current_app.config["APP_NAME"],
        "app_url": current_app.config["APP_URL"],
    }
    send_email(
        subject=f"Welcome to {ctx['app_name']}!",
        recipient=user.email,
        text_body=render_template("email/welcome.txt", **ctx),
        html_body=render_template("email/welcome.html", **ctx),
        background=background,
    )
```

## `auth.py`

```python
import re

from flask import Blueprint, jsonify, request
from sqlalchemy.exc import IntegrityError

from extensions import db
from mailer import send_welcome_email
from models import User, utcnow

bp = Blueprint("auth", __name__, url_prefix="/auth")

EMAIL_RE = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")
MIN_PASSWORD_LEN = 8


@bp.post("/signup")
def signup():
    data = request.get_json(silent=True) or request.form
    email = (data.get("email") or "").strip().lower()
    password = data.get("password") or ""
    name = (data.get("name") or "").strip() or None

    if not EMAIL_RE.match(email):
        return jsonify(error="A valid email address is required."), 400
    if len(password) < MIN_PASSWORD_LEN:
        return jsonify(error=f"Password must be at least {MIN_PASSWORD_LEN} characters."), 400

    user = User(email=email, name=name)
    user.set_password(password)

    db.session.add(user)
    try:
        db.session.commit()
    except IntegrityError:
        db.session.rollback()
        # Don't leak which addresses are registered.
        return jsonify(error="Could not create that account."), 409

    # Only after the account is durably committed. Failures are logged,
    # not surfaced — a dead SMTP server must not fail a valid signup.
    send_welcome_email(user)
    user.welcome_email_sent_at = utcnow()
    db.session.commit()

    return jsonify(id=user.id, email=user.email), 201
```

## `app.py`

```python
import logging
import os

from flask import Flask

from config import Config
from extensions import db


def create_app(config_object=Config):
    app = Flask(__name__)
    app.config.from_object(config_object)

    logging.basicConfig(level=logging.INFO)

    db.init_app(app)

    from auth import bp as auth_bp
    app.register_blueprint(auth_bp)

    with app.app_context():
        db.create_all()  # replace with Flask-Migrate in production

    return app


app = create_app()

if __name__ == "__main__":
    app.run(debug=os.environ.get("FLASK_DEBUG") == "1")
```

## Two things worth knowing

**The background thread is a starter, not a production answer.** It gives you a signup response that doesn't wait on SMTP, but a thread dies with the worker process and never retries — a restart mid-send silently drops the email. When it matters, move `send_welcome_email` behind Celery or RQ: enqueue the `user.id`, re-load and render inside the task, and let the broker handle retries. The `welcome_email_sent_at` column is there so the task can be idempotent (`if user.welcome_email_sent_at: return`) — right now it's just a record, since the thread can't report failure back.

**`EMAIL_RE` is a shape check, not validation.** It rejects obvious garbage and nothing more. If bounce rates matter, add `email-validator` and use `validate_email(email, check_deliverability=True)` in place of the regex.
