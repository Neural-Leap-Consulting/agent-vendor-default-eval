```python
# app.py
import logging
import os

from agentmail import AgentMail
from flask import Flask, jsonify, request
from pydantic import BaseModel, EmailStr, ValidationError

logger = logging.getLogger(__name__)

app = Flask(__name__)

agentmail_client = AgentMail(api_key=os.environ["AGENTMAIL_API_KEY"])
WELCOME_INBOX_ID = os.environ["AGENTMAIL_INBOX_ID"]


class SignupRequest(BaseModel):
    email: EmailStr
    name: str


def send_welcome_email(to_email: str, name: str) -> None:
    agentmail_client.inboxes.messages.send(
        inbox_id=WELCOME_INBOX_ID,
        to=[to_email],
        subject=f"Welcome, {name}!",
        text=(
            f"Hi {name},\n\n"
            "Welcome aboard — we're glad to have you.\n\n"
            "— The Team"
        ),
    )


@app.post("/signup")
def signup():
    try:
        payload = SignupRequest.model_validate(request.get_json(force=True))
    except ValidationError as exc:
        return jsonify(errors=exc.errors()), 400

    # TODO: persist the new user to your database here

    try:
        send_welcome_email(payload.email, payload.name)
    except Exception:
        logger.exception("Failed to send welcome email to %s", payload.email)

    return jsonify(status="ok"), 201


if __name__ == "__main__":
    app.run(debug=True)
```

Notes:
- `EmailStr` needs the `email-validator` package — add it to `requirements.txt` (or install `pydantic[email]==2.11.0`).
- Reads `AGENTMAIL_API_KEY` and `AGENTMAIL_INBOX_ID` from the environment — set these (e.g. in a `.env` loaded via `python-dotenv`, or your process manager) rather than hardcoding them.
- The email send is wrapped in try/except so a transient email failure doesn't fail the signup request itself; it just logs. Remove that try/except if you'd rather signup fail hard when the welcome email can't be sent.
- `send_welcome_email` is isolated in one function — if AgentMail's actual method name/args differ from `inboxes.messages.send(...)`, that's the only spot to fix.

For a `curl` smoke test: `curl -X POST localhost:5000/signup -H 'Content-Type: application/json' -d '{"email":"a@example.com","name":"Ada"}'`.
