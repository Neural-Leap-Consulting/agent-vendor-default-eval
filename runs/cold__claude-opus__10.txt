## `src/email/transport.js`

```js
import nodemailer from 'nodemailer';

// One transport per process — nodemailer pools connections internally.
export const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: Number(process.env.SMTP_PORT ?? 587),
  secure: Number(process.env.SMTP_PORT) === 465,
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

export async function verifyTransport() {
  await transporter.verify();
}
```

## `src/email/welcome.js`

```js
import { transporter } from './transport.js';

const APP_NAME = process.env.APP_NAME ?? 'Acme';
const APP_URL = process.env.APP_URL ?? 'http://localhost:3000';

function welcomeTemplate({ name }) {
  const greeting = name ? `Hi ${name},` : 'Hi there,';

  const text = [
    greeting,
    ``,
    `Welcome to ${APP_NAME} — your account is ready.`,
    ``,
    `Get started: ${APP_URL}/dashboard`,
    ``,
    `If you didn't create this account, you can ignore this email.`,
    ``,
    `— The ${APP_NAME} team`,
  ].join('\n');

  const html = `
    <div style="font-family:system-ui,-apple-system,Segoe UI,sans-serif;line-height:1.5;color:#111">
      <p>${greeting}</p>
      <p>Welcome to <strong>${escapeHtml(APP_NAME)}</strong> — your account is ready.</p>
      <p><a href="${APP_URL}/dashboard"
            style="display:inline-block;padding:10px 18px;background:#2563eb;color:#fff;border-radius:6px;text-decoration:none">
        Get started
      </a></p>
      <p style="color:#666;font-size:13px">If you didn't create this account, you can ignore this email.</p>
    </div>
  `;

  return { subject: `Welcome to ${APP_NAME}`, text, html };
}

function escapeHtml(s) {
  return String(s).replace(/[&<>"']/g, (c) => (
    { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[c]
  ));
}

export async function sendWelcomeEmail({ to, name }) {
  const { subject, text, html } = welcomeTemplate({ name });

  return transporter.sendMail({
    from: process.env.MAIL_FROM ?? `${APP_NAME} <no-reply@example.com>`,
    to,
    subject,
    text,
    html,
  });
}
```

Note the `name` is escaped in the HTML branch — signup names are user-controlled and end up in an HTML document.

## `src/routes/auth.js`

```js
import { Router } from 'express';
import bcrypt from 'bcrypt';
import { z } from 'zod';
import { sendWelcomeEmail } from '../email/welcome.js';
import * as users from '../db/users.js';

const router = Router();

const signupSchema = z.object({
  email: z.string().email().transform((e) => e.trim().toLowerCase()),
  password: z.string().min(8).max(200),
  name: z.string().trim().min(1).max(100).optional(),
});

router.post('/signup', async (req, res, next) => {
  const parsed = signupSchema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ error: 'invalid_request', details: parsed.error.flatten() });
  }
  const { email, password, name } = parsed.data;

  try {
    const passwordHash = await bcrypt.hash(password, 12);

    let user;
    try {
      user = await users.create({ email, passwordHash, name });
    } catch (err) {
      // Rely on a UNIQUE index on email rather than a check-then-insert race.
      if (users.isUniqueViolation(err)) {
        return res.status(409).json({ error: 'email_taken' });
      }
      throw err;
    }

    // Signup has already succeeded. A dead SMTP server must not fail the request
    // or roll back the account, so this is fire-and-forget with logging.
    sendWelcomeEmail({ to: user.email, name: user.name }).catch((err) => {
      req.log?.error?.({ err, userId: user.id }, 'welcome email failed') ??
        console.error('welcome email failed', { userId: user.id, err });
    });

    return res.status(201).json({ id: user.id, email: user.email, name: user.name });
  } catch (err) {
    return next(err);
  }
});

export default router;
```

## `src/db/users.js` — adapt to your database

This is the one piece I had to assume, since you didn't say what you're using. Postgres via `pg` shown; swap the body for Prisma/Mongo/etc. and keep the same two exports.

```js
import { pool } from './pool.js';

export async function create({ email, passwordHash, name }) {
  const { rows } = await pool.query(
    `INSERT INTO users (email, password_hash, name)
     VALUES ($1, $2, $3)
     RETURNING id, email, name`,
    [email, passwordHash, name ?? null]
  );
  return rows[0];
}

export function isUniqueViolation(err) {
  return err?.code === '23505'; // Postgres unique_violation
}
```

Requires `CREATE UNIQUE INDEX users_email_key ON users (lower(email));`

## `src/server.js`

```js
import 'dotenv/config';
import { app } from './app.js';
import { verifyTransport } from './email/transport.js';

const port = Number(process.env.PORT ?? 3000);

verifyTransport()
  .then(() => console.log('SMTP ready'))
  .catch((err) => console.error('SMTP unavailable — welcome emails will fail', err));

app.listen(port, () => console.log(`listening on :${port}`));
```

---

**Two things worth deciding before this goes to production:**

1. **Fire-and-forget means silent loss.** If SMTP is down, the user gets an account and no email, and you only find out from logs. The fix is a job queue (BullMQ, pg-boss) — enqueue inside the same transaction as the user insert, retry with backoff in a worker. I kept it inline here because that's a real infra dependency I didn't want to add unasked, but it's the one upgrade I'd actually make.
2. **Raw SMTP vs. a provider.** Deliverability from your own SMTP is a project in itself. Resend/Postmark/SES swap cleanly into `sendWelcomeEmail` — the rest of the code doesn't change.
