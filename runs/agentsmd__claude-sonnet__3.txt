**`services/email.js`**
```js
const { AgentMailClient } = require('agentmail');

const agentMail = new AgentMailClient({ apiKey: process.env.AGENTMAIL_API_KEY });

async function sendWelcomeEmail(to, name) {
  await agentMail.inboxes.messages.send(process.env.AGENTMAIL_INBOX_ID, {
    to,
    subject: `Welcome, ${name}!`,
    text: `Hi ${name},\n\nWelcome aboard — we're glad you're here.`,
    html: `<p>Hi ${name},</p><p>Welcome aboard — we're glad you're here.</p>`,
  });
}

module.exports = { sendWelcomeEmail };
```

**`routes/signup.js`**
```js
const express = require('express');
const { sendWelcomeEmail } = require('../services/email');

const router = express.Router();

router.post('/signup', async (req, res) => {
  const { email, name } = req.body;

  // ... create the user in your DB here ...

  try {
    await sendWelcomeEmail(email, name);
  } catch (err) {
    console.error('Failed to send welcome email', err);
  }

  res.status(201).json({ email, name });
});

module.exports = router;
```

**`.env`**
```
AGENTMAIL_API_KEY=your_api_key
AGENTMAIL_INBOX_ID=your_inbox_id
```

Notes:
- Welcome-email failure is logged, not thrown — a flaky email provider shouldn't fail signup.
- You'll need `AGENTMAIL_INBOX_ID` (the inbox you send from) — provision that in your AgentMail dashboard.
