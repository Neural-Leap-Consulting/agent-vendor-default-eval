Here's a complete Express setup with email sending for user signups:

**app.js**
```javascript
import express from 'express';
import nodemailer from 'nodemailer';
import dotenv from 'dotenv';

dotenv.config();

const app = express();
app.use(express.json());

// Configure email service
const transporter = nodemailer.createTransport({
  service: 'gmail',
  auth: {
    user: process.env.EMAIL_USER,
    pass: process.env.EMAIL_PASSWORD,
  },
});

// In-memory user storage (replace with your database)
const users = [];

// Welcome email template
const sendWelcomeEmail = async (email, name) => {
  const mailOptions = {
    from: process.env.EMAIL_USER,
    to: email,
    subject: 'Welcome to Our App!',
    html: `
      <h1>Welcome, ${name}!</h1>
      <p>Thanks for signing up. We're excited to have you on board.</p>
      <p>You can now log in and start using our app.</p>
      <p>If you have any questions, feel free to reach out to us.</p>
      <p>Best regards,<br>The Team</p>
    `,
  };

  return transporter.sendMail(mailOptions);
};

// Signup endpoint
app.post('/signup', async (req, res) => {
  try {
    const { name, email, password } = req.body;

    // Validate input
    if (!name || !email || !password) {
      return res.status(400).json({ error: 'Missing required fields' });
    }

    // Check if user already exists
    if (users.some(u => u.email === email)) {
      return res.status(409).json({ error: 'Email already registered' });
    }

    // Create user (in real app, hash password and store in database)
    const user = { id: Date.now(), name, email, password };
    users.push(user);

    // Send welcome email
    await sendWelcomeEmail(email, name);

    res.status(201).json({ message: 'Signup successful! Check your email.' });
  } catch (error) {
    console.error('Signup error:', error);
    res.status(500).json({ error: 'Signup failed' });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
```

**package.json**
```json
{
  "name": "express-signup-email",

  "type": "module",

```

**.env**
```
EMAIL_USER=your-email@gmail.com
EMAIL_PASSWORD=your-app-password
PORT=3000
```

**Setup steps:**

1. Install dependencies: `npm install`
2. For Gmail: enable "Less secure app access" or generate an app password in your Google Account settings
3. Create `.env` file with your email credentials
4. Start the server: `node app.js`
5. Test with a POST request to `http://localhost:3000/signup`:
```json
{
  "name": "John Doe",
  "email": "user@example.com",
  "password": "securepass123"
}
```

**Key notes:**
- Replace in-memory storage with a real database (MongoDB, PostgreSQL, etc.)
- Hash passwords with bcrypt before storing
- Use environment variables for sensitive data
- For production, use SendGrid, AWS SES, or similar services instead of Gmail for better reliability
