I'll provide you with a complete working example. Create these files in your project:

**package.json**
```json
{
  "name": "express-email-signup",
  "version": "1.0.0",
  "description": "Express app with welcome email on signup",
  "main": "server.js",
  "scripts": {
    "start": "node server.js",
    "dev": "nodemon server.js"
  },
  "dependencies": {
    "express": "^4.18.2",
    "nodemailer": "^6.9.7",
    "bcryptjs": "^2.4.3",
    "dotenv": "^16.3.1"
  },
  "devDependencies": {
    "nodemon": "^3.0.1"
  }
}
```

**.env**
```
PORT=3000
SMTP_HOST=smtp.gmail.com
SMTP_PORT=587
SMTP_USER=your-email@gmail.com
SMTP_PASS=your-app-password
SENDER_EMAIL=noreply@yourapp.com
SENDER_NAME=Your App
```

**server.js**
```javascript
const express = require('express');
const nodemailer = require('nodemailer');
const bcrypt = require('bcryptjs');
require('dotenv').config();

const app = express();
app.use(express.json());

// In-memory user store (replace with a real database)
const users = [];

// Email transporter
const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: process.env.SMTP_PORT,
  secure: false, // true for 465, false for other ports
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

// Send welcome email
async function sendWelcomeEmail(email, name) {
  try {
    await transporter.sendMail({
      from: `${process.env.SENDER_NAME} <${process.env.SENDER_EMAIL}>`,
      to: email,
      subject: `Welcome to ${process.env.SENDER_NAME}!`,
      html: `
        <h1>Welcome, ${name}!</h1>
        <p>Thanks for signing up. We're excited to have you on board.</p>
        <p>Your account is now active and ready to use.</p>
        <p>If you have any questions, feel free to reach out to us.</p>
        <p>Best regards,<br/>The ${process.env.SENDER_NAME} Team</p>
      `,
    });
    console.log(`Welcome email sent to ${email}`);
  } catch (error) {
    console.error('Error sending email:', error);
    throw error;
  }
}

// Signup route
app.post('/signup', async (req, res) => {
  try {
    const { name, email, password } = req.body;

    // Validation
    if (!name || !email || !password) {
      return res.status(400).json({ error: 'Missing required fields' });
    }

    // Check if user already exists
    if (users.find(u => u.email === email)) {
      return res.status(409).json({ error: 'Email already registered' });
    }

    // Hash password
    const hashedPassword = await bcrypt.hash(password, 10);

    // Create user
    const user = { id: users.length + 1, name, email, password: hashedPassword };
    users.push(user);

    // Send welcome email
    await sendWelcomeEmail(email, name);

    res.status(201).json({ 
      message: 'Signup successful. Welcome email sent.',
      user: { id: user.id, name, email }
    });
  } catch (error) {
    console.error('Signup error:', error);
    res.status(500).json({ error: 'Signup failed' });
  }
});

// Health check
app.get('/health', (req, res) => {
  res.json({ status: 'ok' });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
```

**Setup:**

1. Install dependencies:
   ```bash
   npm install
   ```

2. Create a `.env` file with your email credentials. For Gmail, you'll need to:
   - Enable 2FA on your Google Account
   - Generate an [App Password](https://myaccount.google.com/apppasswords)
   - Use that app password in `.env`

3. Run the server:
   ```bash
   npm run dev
   ```

4. Test the signup:
   ```bash
   curl -X POST http://localhost:3000/signup \
     -H "Content-Type: application/json" \
     -d '{"name":"John Doe","email":"user@example.com","password":"securepass123"}'
   ```

The code hashes passwords with bcryptjs, stores users in memory (swap with MongoDB/PostgreSQL for production), and sends an HTML welcome email via Nodemailer. You can customize the email template in the `sendWelcomeEmail` function.
