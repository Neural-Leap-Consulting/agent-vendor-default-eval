I don't have filesystem tools available in this session, so here's the code to add to your project directly.

**Install dependency:**
```bash
npm install nodemailer
```

**`config/mailer.js`** — SMTP transporter, configured from env vars:

```js
const nodemailer = require('nodemailer');

const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: Number(process.env.SMTP_PORT),
  secure: process.env.SMTP_PORT === '465',
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

module.exports = transporter;
```

**`emails/sendWelcomeEmail.js`**:

```js
const transporter = require('../config/mailer');

async function sendWelcomeEmail(user) {
  await transporter.sendMail({
    from: process.env.EMAIL_FROM,
    to: user.email,
    subject: 'Welcome!',
    html: `<p>Hi ${user.name}, welcome aboard!</p>`,
  });
}

module.exports = sendWelcomeEmail;
```

**`routes/auth.js`** — signup route, wired in:

```js
const express = require('express');
const router = express.Router();
const sendWelcomeEmail = require('../emails/sendWelcomeEmail');
// const User = require('../models/User'); // your existing user model

router.post('/signup', async (req, res) => {
  const { name, email, password } = req.body;

  const user = await User.create({ name, email, password });

  try {
    await sendWelcomeEmail(user);
  } catch (err) {
    console.error('Failed to send welcome email:', err);
  }

  res.status(201).json({ id: user.id, name: user.name, email: user.email });
});

module.exports = router;
```

**`.env`** additions:

```
SMTP_HOST=smtp.example.com
SMTP_PORT=587
SMTP_USER=your-smtp-username
SMTP_PASS=your-smtp-password
EMAIL_FROM="Your App <no-reply@yourapp.com>"
```

Notes:
- The welcome email send is wrapped in its own try/catch so a mail-provider hiccup doesn't fail the signup response — swap the `console.error` for whatever logging you use.
- Swap in your real user-creation call where `User.create(...)` is; I left it as a placeholder since I don't know your data layer.
- For a transactional-email provider (SendGrid, Postmark, SES, etc.) instead of raw SMTP, only `config/mailer.js` changes — happy to adjust if you tell me which one you're using.
