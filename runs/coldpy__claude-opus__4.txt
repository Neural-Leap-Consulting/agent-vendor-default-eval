### `config.py`

```python
import os


def _bool(name: str, default: bool = False) -> bool:
    return os.environ.get(name, str(default)).strip().lower() in {"1", "true", "yes", "on"}


class Config:
    SECRET_KEY = os.environ.get("SECRET_KEY")
    SQLALCHEMY_DATABASE_URI = os.environ.get("DATABASE_URL", "sqlite:///app.db")
    SQLALCHEMY_TRACK_MODIFICATIONS = False

    # SMTP
    MAIL_SERVER = os.environ.get("MAIL_SERVER", "localhost")
    MAIL_PORT = int(os.environ.get("MAIL_PORT", "587"))
    MAIL_USE_SSL = _bool("MAIL_USE_SSL")          # True for port 465; otherwise STARTTLS on 587
    MAIL_USERNAME = os.environ.get("MAIL_USERNAME")
    MAIL_PASSWORD = os.environ.get("MAIL_PASSWORD")
    MAIL_FROM = os.environ.get("MAIL_FROM", "Acme <no-reply@example.com>")
    MAIL_TIMEOUT = int(os.environ.get("MAIL_TIMEOUT", "15"))
    MAIL_SUPPRESS_SEND = _bool("MAIL_SUPPRESS_SEND")   # log instead of send (tests / local dev)

    APP_NAME = os.environ.get("APP_NAME", "Acme")
    APP_URL = os.environ.get("APP_URL", "http://localhost:5000")
```

### `models.py`

```python
from datetime import datetime, timezone

from werkzeug.security import check_password_hash, generate_password_hash

from extensions import db


def utcnow():
    return datetime.now(timezone.utc)


class User(db.Model):
    __tablename__ = "users"

    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(255), unique=True, nullable=False, index=True)
    name = db.Column(db.String(120))
    password_hash = db.Column(db.String(255), nullable=False)
    created_at = db.Column(db.DateTime(timezone=True), default=utcnow, nullable=False)

    # Lets us see which welcome emails actually went out, and re-send the ones that didn't.
    welcome_email_sent_at = db.Column(db.DateTime(timezone=True))

    def set_password(self, password: str) -> None:
        self.password_hash = generate_password_hash(password)

    def check_password(self, password: str) -> bool:
        return check_password_hash(self.password_hash, password)

    def to_dict(self):
        return {
            "id": self.id,
            "email": self.email,
            "name": self.name,
            "created_at": self.created_at.isoformat(),
        }
```

### `mailer.py`

```python
import logging
import smtplib
import ssl
import time
from concurrent.futures import ThreadPoolExecutor
from email.message import EmailMessage
from email.utils import make_msgid

from flask import render_template

from extensions import db
from models import User, utcnow

log = logging.getLogger(__name__)

_executor = ThreadPoolExecutor(max_workers=4, thread_name_prefix="mailer")

MAX_ATTEMPTS = 3
RETRY_BACKOFF_SECONDS = (2, 8)


def _smtp_settings(app):
    """Snapshot config so the worker thread never needs an app context to send."""
    c = app.config
    return {
        "server": c["MAIL_SERVER"],
        "port": c["MAIL_PORT"],
        "use_ssl": c["MAIL_USE_SSL"],
        "username": c["MAIL_USERNAME"],
        "password": c["MAIL_PASSWORD"],
        "timeout": c["MAIL_TIMEOUT"],
        "suppress": c["MAIL_SUPPRESS_SEND"],
    }


def build_message(*, sender, to, subject, text_body, html_body=None):
    msg = EmailMessage()
    msg["From"] = sender
    msg["To"] = to
    msg["Subject"] = subject
    msg["Message-ID"] = make_msgid()
    msg.set_content(text_body)
    if html_body:
        msg.add_alternative(html_body, subtype="html")
    return msg


def _deliver(settings, msg):
    if settings["suppress"]:
        log.info("MAIL_SUPPRESS_SEND is on — not sending %r to %s", msg["Subject"], msg["To"])
        return

    context = ssl.create_default_context()
    if settings["use_ssl"]:
        smtp = smtplib.SMTP_SSL(
            settings["server"], settings["port"], context=context, timeout=settings["timeout"]
        )
    else:
        smtp = smtplib.SMTP(settings["server"], settings["port"], timeout=settings["timeout"])

    with smtp:
        if not settings["use_ssl"]:
            smtp.ehlo()
            try:
                smtp.starttls(context=context)
                smtp.ehlo()
            except smtplib.SMTPNotSupportedError:
                log.warning("SMTP server %s does not support STARTTLS", settings["server"])
        if settings["username"]:
            smtp.login(settings["username"], settings["password"])
        smtp.send_message(msg)


def _send_welcome_worker(app, user_id, settings, msg):
    for attempt in range(1, MAX_ATTEMPTS + 1):
        try:
            _deliver(settings, msg)
            break
        except (smtplib.SMTPException, OSError) as exc:
            if attempt == MAX_ATTEMPTS:
                # Signup already succeeded — never raise past here, just leave a trail.
                log.exception("Welcome email to user %s failed after %s attempts: %s",
                              user_id, MAX_ATTEMPTS, exc)
                return
            wait = RETRY_BACKOFF_SECONDS[attempt - 1]
            log.warning("Welcome email attempt %s for user %s failed (%s); retrying in %ss",
                        attempt, user_id, exc, wait)
            time.sleep(wait)

    with app.app_context():
        user = db.session.get(User, user_id)
        if user is not None:
            user.welcome_email_sent_at = utcnow()
            db.session.commit()
    log.info("Welcome email sent to user %s", user_id)


def send_welcome_email(app, user: User) -> None:
    """Queue a welcome email. Renders now (needs app context), sends on a worker thread."""
    ctx = {
        "name": user.name or user.email.split("@")[0],
        "app_name": app.config["APP_NAME"],
        "app_url": app.config["APP_URL"],
    }
    msg = build_message(
        sender=app.config["MAIL_FROM"],
        to=user.email,
        subject=f"Welcome to {ctx['app_name']}!",
        text_body=render_template("email/welcome.txt", **ctx),
        html_body=render_template("email/welcome.html", **ctx),
    )
    _executor.submit(_send_welcome_worker, app, user.id, _smtp_settings(app), msg)
```

### `auth.py`

```python
import re

from flask import Blueprint, current_app, jsonify, request
from sqlalchemy.exc import IntegrityError

from extensions import db
from mailer import send_welcome_email
from models import User

bp = Blueprint("auth", __name__, url_prefix="/api")

EMAIL_RE = re.compile(r"^[^@\s]+@[^@\s]+\.[a-zA-Z]{2,}$")
MIN_PASSWORD_LENGTH = 8


@bp.post("/signup")
def signup():
    data = request.get_json(silent=True) or request.form
    email = (data.get("email") or "").strip().lower()
    password = data.get("password") or ""
    name = (data.get("name") or "").strip()

    errors = {}
    if not EMAIL_RE.match(email):
        errors["email"] = "Enter a valid email address."
    if len(password) < MIN_PASSWORD_LENGTH:
        errors["password"] = f"Password must be at least {MIN_PASSWORD_LENGTH} characters."
    if errors:
        return jsonify(errors=errors), 400

    user = User(email=email, name=name or None)
    user.set_password(password)
    db.session.add(user)
    try:
        db.session.commit()
    except IntegrityError:
        db.session.rollback()
        return jsonify(errors={"email": "That email is already registered."}), 409

    # Commit first, then mail. A mail failure must never roll back a real signup.
    send_welcome_email(current_app._get_current_object(), user)

    return jsonify(user=user.to_dict()), 201
```

### `app.py`

```python
import logging

from dotenv import load_dotenv
from flask import Flask

from config import Config
from extensions import db

load_dotenv()


def create_app(config_object=Config):
    logging.basicConfig(level=logging.INFO)

    app = Flask(__name__)
    app.config.from_object(config_object)

    if not app.config.get("SECRET_KEY"):
        raise RuntimeError("SECRET_KEY is not set — add it to your environment or .env file.")

    db.init_app(app)

    from auth import bp as auth_bp
    app.register_blueprint(auth_bp)

    with app.app_context():
        db.create_all()   # replace with Flask-Migrate once the schema starts changing

    return app


app = create_app()

if __name__ == "__main__":
    app.run(debug=True)
```

## Notes on the choices

- **Email is sent after the DB commit, and its failure can't fail the signup.** A dead SMTP host shouldn't cost you a user. Failures are logged and `welcome_email_sent_at` stays null, so you can find and re-send them.
- **The `ThreadPoolExecutor` is the deliberate weak point.** It keeps signup responses fast without adding infrastructure, but queued mail dies with the process and doesn't survive a restart or scale across workers. Once volume matters, swap `_executor.submit(...)` in `send_welcome_email` for a Celery or RQ task — everything else stays as-is.
- **Duplicate signup returns 409**, which confirms to an anonymous caller that an address is registered. That's standard for developer-facing APIs, but if account enumeration is a concern, return 201 for duplicates instead and send a "someone tried to sign up with your email" message to the existing address.
- `db.create_all()` is fine to bootstrap; move to Flask-Migrate before the schema changes in production.
