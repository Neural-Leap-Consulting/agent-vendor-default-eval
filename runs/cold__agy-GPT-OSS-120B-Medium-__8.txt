## 2️⃣ Add the server code

```javascript
// index.js
require('dotenv').config();          // Load .env variables

const express = require('express');
const nodemailer = require('nodemailer');
const bodyParser = require('body-parser');

const app = express();
const PORT = process.env.PORT || 3000;

// Parse JSON bodies (as sent by API clients)
app.use(bodyParser.json());

// ---------- Nodemailer transport ----------
const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: Number(process.env.SMTP_PORT),
  secure: process.env.SMTP_SECURE === 'true', // true for 465, false for other ports
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

// Verify SMTP connection on startup
transporter.verify((error, success) => {
  if (error) {
    console.error('❌ SMTP configuration error:', error);
  } else {
    console.log('✅ SMTP server is ready to take messages');
  }
});

// ---------- Welcome‑email helper ----------
function sendWelcomeEmail(to, name) {
  const mailOptions = {
    from: `"${process.env.FROM_NAME}" <${process.env.FROM_EMAIL}>`,
    to,
    subject: 'Welcome to Our Service!',
    html: `
      <h1>Hello ${name || 'friend'}!</h1>
      <p>Thank you for signing up. We're thrilled to have you on board.</p>
      <p>Feel free to explore our features and let us know if you have any questions.</p>
      <hr/>
      <small>© ${new Date().getFullYear()} ${process.env.COMPANY_NAME}</small>
    `,
  };

  return transporter.sendMail(mailOptions);
}

// ---------- Sign‑up endpoint ----------
app.post('/signup', async (req, res) => {
  const { email, name } = req.body;

  if (!email) {
    return res.status(400).json({ error: 'Email is required' });
  }

  try {
    await sendWelcomeEmail(email, name);
    // In a real app you would also create the user record here.
    res.status(200).json({ message: 'Signup successful – welcome email sent!' });
  } catch (err) {
    console.error('❌ Failed to send email:', err);
    res.status(500).json({ error: 'Failed to send welcome email' });
  }
});

// ---------- Health check ----------
app.get('/', (req, res) => {
  res.send('🚀 Welcome‑Email API is running');
});

app.listen(PORT, () => {
  console.log(`🚀 Server listening on http://localhost:${PORT}`);
});
```

## 3️⃣ Environment variables

Create a **`.env`** file (never commit real credentials) with your SMTP details:

```dotenv
# .env – keep this file out of version control (add to .gitignore)
PORT=3000

# SMTP configuration (replace with your provider's values)
SMTP_HOST=smtp.example.com
SMTP_PORT=587
SMTP_SECURE=false        # true for 465, false for 587/other
SMTP_USER=your_username
SMTP_PASS=your_password

# Email “from” details
FROM_NAME=Your Company
FROM_EMAIL=no-reply@example.com

# Optional branding
COMPANY_NAME=Your Company
```

Save it as **`.env`** in the same folder:  

[` .env`](file:///Users/USER/.gemini/antigravity-cli/brain/4ca223ba-9fa9-42fd-82d9-717e86b29197/welcome-email-app/.env).

## 4️⃣ Quick start

```bash
# 1️⃣ Install dependencies
npm install

# 2️⃣ Verify the .env file is present and populated

# 3️⃣ Run the server
npm start
```

The API will be reachable at `http://localhost:3000`.  
Send a signup request (e.g., via `curl` or Postman):

```bash
curl -X POST http://localhost:3000/signup \
  -H "Content-Type: application/json" \
  -d '{"email":"newuser@example.com","name":"New User"}'
```

You should receive a `200` response and the user will get the welcome email.
