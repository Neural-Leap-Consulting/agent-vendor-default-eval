### `config.py`

```python
import os


def _bool(name, default=False):
    return os.environ.get(name, str(default)).lower() in ("1", "true", "yes")


class Config:
    SECRET_KEY = os.environ.get("SECRET_KEY", "dev-only-not-secret")
    SQLALCHEMY_DATABASE_URI = os.environ.get("DATABASE_URL", "sqlite:///app.db")
    SQLALCHEMY_TRACK_MODIFICATIONS = False

    MAIL_BACKEND = os.environ.get("MAIL_BACKEND", "console")  # console | smtp
    MAIL_SERVER = os.environ.get("MAIL_SERVER", "localhost")
    MAIL_PORT = int(os.environ.get("MAIL_PORT", 587))
    MAIL_USE_TLS = _bool("MAIL_USE_TLS", True)     # STARTTLS, port 587
    MAIL_USE_SSL = _bool("MAIL_USE_SSL", False)    # implicit TLS, port 465
    MAIL_USERNAME = os.environ.get("MAIL_USERNAME")
    MAIL_PASSWORD = os.environ.get("MAIL_PASSWORD")
    MAIL_FROM = os.environ.get("MAIL_FROM", "no-reply@localhost")
    MAIL_TIMEOUT = int(os.environ.get("MAIL_TIMEOUT", 10))

    APP_NAME = os.environ.get("APP_NAME", "Acme")
    APP_URL = os.environ.get("APP_URL", "http://localhost:5000")
```

### `app/models.py`

```python
from datetime import datetime, timezone

from flask_sqlalchemy import SQLAlchemy
from werkzeug.security import check_password_hash, generate_password_hash

db = SQLAlchemy()


class User(db.Model):
    __tablename__ = "users"

    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(255), unique=True, nullable=False, index=True)
    name = db.Column(db.String(120))
    password_hash = db.Column(db.String(255), nullable=False)
    welcome_email_sent_at = db.Column(db.DateTime(timezone=True))
    created_at = db.Column(
        db.DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
    )

    def set_password(self, raw):
        self.password_hash = generate_password_hash(raw)

    def check_password(self, raw):
        return check_password_hash(self.password_hash, raw)
```

### `app/mailer.py`

The important part: the send runs on a background thread, so a slow or dead SMTP server can't stall the signup request, and a delivery failure can't roll back an account that was already created.

```python
import logging
import smtplib
import threading
from email.message import EmailMessage
from email.utils import make_msgid

from flask import current_app, render_template

log = logging.getLogger(__name__)


def _build_message(cfg, *, to, subject, text_body, html_body):
    msg = EmailMessage()
    msg["Subject"] = subject
    msg["From"] = cfg["MAIL_FROM"]
    msg["To"] = to
    msg["Message-ID"] = make_msgid()
    msg.set_content(text_body)
    msg.add_alternative(html_body, subtype="html")
    return msg


def _deliver(cfg, msg):
    """Blocking send. Runs on a worker thread."""
    if cfg["MAIL_BACKEND"] == "console":
        print("--- EMAIL ---\n" + msg.as_string() + "\n--- END EMAIL ---", flush=True)
        return

    if cfg["MAIL_USE_SSL"]:
        smtp = smtplib.SMTP_SSL(
            cfg["MAIL_SERVER"], cfg["MAIL_PORT"], timeout=cfg["MAIL_TIMEOUT"]
        )
    else:
        smtp = smtplib.SMTP(
            cfg["MAIL_SERVER"], cfg["MAIL_PORT"], timeout=cfg["MAIL_TIMEOUT"]
        )
    with smtp:
        smtp.ehlo()
        if cfg["MAIL_USE_TLS"] and not cfg["MAIL_USE_SSL"]:
            smtp.starttls()
            smtp.ehlo()
        if cfg["MAIL_USERNAME"]:
            smtp.login(cfg["MAIL_USERNAME"], cfg["MAIL_PASSWORD"])
        smtp.send_message(msg)


def send_email(to, subject, template, **context):
    """Render `templates/email/<template>.{txt,html}` and send it in the background."""
    app = current_app._get_current_object()
    cfg = {
        k: app.config[k]
        for k in (
            "MAIL_BACKEND", "MAIL_SERVER", "MAIL_PORT", "MAIL_USE_TLS",
            "MAIL_USE_SSL", "MAIL_USERNAME", "MAIL_PASSWORD",
            "MAIL_FROM", "MAIL_TIMEOUT",
        )
    }

    # Render inside the request context — templates need app config/url helpers.
    text_body = render_template(f"email/{template}.txt", **context)
    html_body = render_template(f"email/{template}.html", **context)
    msg = _build_message(
        cfg, to=to, subject=subject, text_body=text_body, html_body=html_body
    )

    if app.config.get("MAIL_SEND_SYNCHRONOUSLY"):  # tests
        _deliver(cfg, msg)
        return

    def run():
        try:
            _deliver(cfg, msg)
        except Exception:
            log.exception("Failed to send %r email to %s", template, to)

    threading.Thread(target=run, daemon=True, name=f"mail-{template}").start()


def send_welcome_email(user):
    send_email(
        to=user.email,
        subject=f"Welcome to {current_app.config['APP_NAME']}!",
        template="welcome",
        user=user,
        app_name=current_app.config["APP_NAME"],
        app_url=current_app.config["APP_URL"],
    )
```

### `app/auth.py`

```python
from datetime import datetime, timezone

from email_validator import EmailNotValidError, validate_email
from flask import Blueprint, jsonify, request
from sqlalchemy.exc import IntegrityError

from .mailer import send_welcome_email
from .models import User, db

bp = Blueprint("auth", __name__)

MIN_PASSWORD_LENGTH = 8


@bp.post("/signup")
def signup():
    data = request.get_json(silent=True) or {}
    email = (data.get("email") or "").strip()
    password = data.get("password") or ""
    name = (data.get("name") or "").strip() or None

    try:
        email = validate_email(email, check_deliverability=False).normalized.lower()
    except EmailNotValidError as exc:
        return jsonify(error=str(exc)), 400

    if len(password) < MIN_PASSWORD_LENGTH:
        return jsonify(
            error=f"Password must be at least {MIN_PASSWORD_LENGTH} characters."
        ), 400

    user = User(email=email, name=name)
    user.set_password(password)
    db.session.add(user)
    try:
        db.session.commit()
    except IntegrityError:
        # Relies on the unique index, so concurrent signups can't both win.
        db.session.rollback()
        return jsonify(error="An account with that email already exists."), 409

    # Account is committed; email delivery is best-effort and must not fail signup.
    user.welcome_email_sent_at = datetime.now(timezone.utc)
    db.session.commit()
    send_welcome_email(user)

    return jsonify(id=user.id, email=user.email, name=user.name), 201
```

### `app/__init__.py`

```python
import logging

from dotenv import load_dotenv
from flask import Flask

from config import Config
from .models import db


def create_app(config_object=Config):
    load_dotenv()
    logging.basicConfig(level=logging.INFO)

    app = Flask(__name__)
    app.config.from_object(config_object)

    db.init_app(app)

    from .auth import bp as auth_bp
    app.register_blueprint(auth_bp)

    with app.app_context():
        db.create_all()  # swap for Flask-Migrate in production

    return app
```

## Two things to know before this goes to production

**The background thread is a pragmatic default, not a durable one.** If the process is killed between the commit and the send, that welcome email is simply lost — there's no retry and no record of the attempt beyond `welcome_email_sent_at`, which this code sets optimistically. That's usually acceptable for a welcome email and never acceptable for a password reset. When you outgrow it, move `send_welcome_email` behind Celery or RQ; the call site in `auth.py` doesn't change, only `mailer.py` does.

**Raw SMTP works but a provider API is better at scale.** SendGrid, Postmark, and SES give you bounce handling, suppression lists, and deliverability reporting that `smtplib` can't. Swapping `_deliver()` for their SDK is a contained change — everything above it stays the same.
