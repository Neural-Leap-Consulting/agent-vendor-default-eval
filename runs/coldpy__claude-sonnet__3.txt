## `app.py`

```python
import os
import threading
from flask import Flask, request, jsonify
from flask_sqlalchemy import SQLAlchemy
from flask_mail import Mail, Message
from werkzeug.security import generate_password_hash
from dotenv import load_dotenv

load_dotenv()

app = Flask(__name__)
app.config["SECRET_KEY"] = os.environ["SECRET_KEY"]
app.config["SQLALCHEMY_DATABASE_URI"] = os.environ.get("DATABASE_URL", "sqlite:///app.db")
app.config["MAIL_SERVER"] = os.environ["MAIL_SERVER"]
app.config["MAIL_PORT"] = int(os.environ.get("MAIL_PORT", 587))
app.config["MAIL_USE_TLS"] = os.environ.get("MAIL_USE_TLS", "true").lower() == "true"
app.config["MAIL_USERNAME"] = os.environ["MAIL_USERNAME"]
app.config["MAIL_PASSWORD"] = os.environ["MAIL_PASSWORD"]
app.config["MAIL_DEFAULT_SENDER"] = os.environ["MAIL_DEFAULT_SENDER"]

db = SQLAlchemy(app)
mail = Mail(app)


class User(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(255), unique=True, nullable=False)
    password_hash = db.Column(db.String(255), nullable=False)


def send_welcome_email(recipient_email):
    msg = Message(
        subject="Welcome!",
        recipients=[recipient_email],
        body=f"Hi,\n\nThanks for signing up — we're glad to have you.\n\nThe Team",
    )
    with app.app_context():
        mail.send(msg)


@app.route("/signup", methods=["POST"])
def signup():
    data = request.get_json(silent=True) or {}
    email = data.get("email", "").strip().lower()
    password = data.get("password", "")

    if not email or not password:
        return jsonify({"error": "email and password are required"}), 400

    if User.query.filter_by(email=email).first():
        return jsonify({"error": "email already registered"}), 409

    user = User(email=email, password_hash=generate_password_hash(password))
    db.session.add(user)
    db.session.commit()

    # Send off the thread so the request doesn't block on SMTP latency.
    threading.Thread(target=send_welcome_email, args=(email,), daemon=True).start()

    return jsonify({"id": user.id, "email": user.email}), 201


if __name__ == "__main__":
    with app.app_context():
        db.create_all()
    app.run(debug=True)
```

Notes:
- Uses `Flask-Mail` over SMTP; swap the `MAIL_*` env vars for your provider (Gmail needs an [app password](https://myaccount.google.com/apppasswords), or use SendGrid/Mailgun/SES for production — more reliable deliverability).
- Email is sent in a background thread so signup doesn't wait on SMTP. For real production traffic, prefer a task queue (Celery/RQ) over a bare thread so failures retry instead of silently vanishing.
- Password is hashed with `werkzeug.security`; adjust the `User` model to fit your existing schema if you already have one.

Let me know if you're using a specific email provider (SendGrid, SES, Mailgun) or want this wired into an existing project structure — I can adjust the code accordingly.
