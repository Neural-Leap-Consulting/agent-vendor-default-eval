**`src/mailer.js`** — transport + the send function
```js
import nodemailer from 'nodemailer';

const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: Number(process.env.SMTP_PORT ?? 587),
  secure: Number(process.env.SMTP_PORT) === 465, // 465 = implicit TLS, 587 = STARTTLS
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

// Fail loudly at boot rather than on the first signup.
export async function verifyMailer() {
  await transporter.verify();
}

export function sendMail({ to, subject, text, html }) {
  return transporter.sendMail({
    from: process.env.MAIL_FROM,
    to,
    subject,
    text,
    html,
  });
}
```

import bcrypt from 'bcrypt';

const usersByEmail = new Map();

export async function findByEmail(email) {
  return usersByEmail.get(email) ?? null;
}

export async function create({ email, name, password }) {
  const user = {
    id: crypto.randomUUID(),
    email,
    name,
    passwordHash: await bcrypt.hash(password, 12),
    createdAt: new Date(),
    welcomeEmailSentAt: null,
  };
  usersByEmail.set(email, user);
  return user;
}

export async function markWelcomeEmailSent(user) {
  user.welcomeEmailSentAt = new Date();

**`src/routes/auth.js`** — the signup route
```js
import express from 'express';
import * as users from '../users.js';
import { sendMail } from '../mailer.js';
import { welcomeEmail } from '../emails/welcome.js';

export const authRouter = express.Router();

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

authRouter.post('/signup', async (req, res, next) => {
  try {
    const email = String(req.body?.email ?? '').trim().toLowerCase();
    const name = String(req.body?.name ?? '').trim() || null;
    const password = String(req.body?.password ?? '');

    if (!EMAIL_RE.test(email)) {
      return res.status(400).json({ error: 'A valid email is required.' });
    }
    if (password.length < 8) {
      return res.status(400).json({ error: 'Password must be at least 8 characters.' });
    }
    if (await users.findByEmail(email)) {
      return res.status(409).json({ error: 'That email is already registered.' });
    }

    const user = await users.create({ email, name, password });

    // Respond first: signup must not fail or stall because the mail server is slow.
    res.status(201).json({ id: user.id, email: user.email, name: user.name });

    // Best-effort send after the response is out the door.
    deliverWelcomeEmail(user).catch((err) => {
      console.error('welcome email failed', { userId: user.id, err });
    });
  } catch (err) {
    next(err);
  }
});

async function deliverWelcomeEmail(user) {
  if (user.welcomeEmailSentAt) return; // never double-send

  const { subject, text, html } = welcomeEmail({
    name: user.name,
    appName: process.env.APP_NAME ?? 'our app',
    appUrl: process.env.APP_URL ?? 'https://example.com',
  });

  await sendMail({ to: user.email, subject, text, html });
  await users.markWelcomeEmailSent(user);
}
```

import express from 'express';
import { authRouter } from './routes/auth.js';

export function createApp() {
  const app = express();
  app.use(express.json());
  app.use('/auth', authRouter);

  app.use((err, _req, res, _next) => {
    console.error(err);
    res.status(500).json({ error: 'Something went wrong.' });
  });

  return app;
}

## Two decisions worth knowing about

**The email is sent after the response, and a failure doesn't fail the signup.** A user who successfully created an account shouldn't see a 500 because your mail provider hiccuped. The tradeoff: if the process dies between the response and the send, that email is silently lost — `welcomeEmailSentAt` stays null but nothing retries it.

**That's the piece to upgrade before production.** Replace the `deliverWelcomeEmail(user).catch(...)` line with an enqueue onto a real job queue (BullMQ/Redis, or `pgboss` if you're already on Postgres) so sends get retries with backoff and a dead-letter path. The function bodies stay exactly as written — only the trigger moves. `welcomeEmailSentAt` is already there to keep the job idempotent across retries.

I left the user store as an in-memory `Map` since you didn't mention a database — `src/users.js` is the only file that needs replacing when you wire up the real one.
