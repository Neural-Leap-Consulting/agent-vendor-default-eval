**`.env.example`**
```
AGENTMAIL_API_KEY=
AGENTMAIL_INBOX_ID=welcome@yourdomain.agentmail.to
PORT=3000
```

**`src/services/agentmailClient.js`**
```js
const { AgentMailClient } = require("agentmail");

if (!process.env.AGENTMAIL_API_KEY) {
  throw new Error("AGENTMAIL_API_KEY is not set");
}

const agentmail = new AgentMailClient({
  apiKey: process.env.AGENTMAIL_API_KEY,
});

module.exports = { agentmail };
```

**`src/services/emailService.js`**
```js
const { agentmail } = require("./agentmailClient");

const WELCOME_INBOX_ID = process.env.AGENTMAIL_INBOX_ID;

async function sendWelcomeEmail(toEmail, name) {
  await agentmail.inboxes.messages.send(WELCOME_INBOX_ID, {
    to: [toEmail],
    subject: "Welcome!",
    text: `Hi ${name},\n\nWelcome aboard — glad to have you.\n\nThanks,\nThe Team`,
    html: `<p>Hi ${name},</p><p>Welcome aboard — glad to have you.</p><p>Thanks,<br/>The Team</p>`,
  });
}

module.exports = { sendWelcomeEmail };
```

**`src/routes/signup.js`**
```js
const express = require("express");
const { sendWelcomeEmail } = require("../services/emailService");

const router = express.Router();

router.post("/signup", async (req, res) => {
  const { email, name } = req.body;

  if (!email || !name) {
    return res.status(400).json({ error: "email and name are required" });
  }

  // TODO: replace with your real user creation logic
  const user = { email, name };

  try {
    await sendWelcomeEmail(user.email, user.name);
  } catch (err) {
    console.error("Failed to send welcome email:", err);
  }

  res.status(201).json({ user });
});

module.exports = router;
```

**`src/app.js`**
```js
require("dotenv").config();
const express = require("express");
const signupRouter = require("./routes/signup");

const app = express();
app.use(express.json());
app.use(signupRouter);

const port = process.env.PORT || 3000;
app.listen(port, () => console.log(`Listening on port ${port}`));
```

- Signup succeeds even if the welcome email fails (logged, not thrown) — adjust if you want signup to hard-fail on email errors.
