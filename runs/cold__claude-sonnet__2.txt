No file tools are available to me in this session, so here's the code to add directly to your project.

**1. Install nodemailer**
```bash
npm install nodemailer
```

**2. `services/emailService.js`**
```js
const nodemailer = require('nodemailer');

const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: Number(process.env.SMTP_PORT),
  secure: process.env.SMTP_PORT === '465',
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

async function sendWelcomeEmail(to, name) {
  await transporter.sendMail({
    from: process.env.EMAIL_FROM,
    to,
    subject: 'Welcome!',
    html: `<p>Hi ${name},</p><p>Thanks for signing up. We're glad to have you.</p>`,
  });
}

module.exports = { sendWelcomeEmail };
```

**3. Signup route** (e.g. `routes/auth.js`)
```js
const express = require('express');
const router = express.Router();
const { sendWelcomeEmail } = require('../services/emailService');

router.post('/signup', async (req, res) => {
  const { email, name, password } = req.body;

  // ... create user in your database here ...

  res.status(201).json({ message: 'Signup successful' });

  sendWelcomeEmail(email, name).catch((err) => {
    console.error('Failed to send welcome email:', err);
  });
});

module.exports = router;
```

**4. `.env`**
```
SMTP_HOST=smtp.example.com
SMTP_PORT=587
SMTP_USER=your_smtp_user
SMTP_PASS=your_smtp_pass
EMAIL_FROM="Your App <no-reply@yourapp.com>"
```

Key design choices:
- The email send happens *after* the response is sent, and its failure is caught and logged rather than failing the signup — a flaky email provider shouldn't block account creation.
- Swap the SMTP transport for a provider SDK (SendGrid, Postmark, SES, etc.) if you're using one — same `sendWelcomeEmail` interface.

For production, consider a queue (e.g. BullMQ) instead of fire-and-forget so failed sends can retry — let me know if you want that added.
