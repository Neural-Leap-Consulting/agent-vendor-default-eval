Here’s a minimal Flask service that creates a user – via a POST to **/signup** – and sends a welcome email using **AgentMail**:

```python
import os
from flask import Flask, request, jsonify
from agentmail import Mailer

app = Flask(__name__)

# AgentMail API key must be set in the environment
MAILER = Mailer(api_key=os.getenv('AGENTMAIL_API_KEY'))

WELCOME_SUBJECT = "Welcome to Our Service!"
WELCOME_BODY_TEMPLATE = """Hello {name},

Thank you for signing up! We're excited to have you on board.

Best regards,
The Team"""

@app.route('/signup', methods=['POST'])
def signup():
    data = request.get_json()
    email = data.get('email')
    name = data.get('name', 'User')
    if not email:
        return jsonify({"error": "Email is required"}), 400

    # TODO: Add user‑creation logic (e.g. DB insert) here

    try:
        MAILER.send_email(
            to=email,
            subject=WELCOME_SUBJECT,
            body=WELCOME_BODY_TEMPLATE.format(name=name)
        )
    except Exception as e:
        # In a real app you’d log the exception
        return jsonify({"error": f"Failed to send welcome email: {e}"}), 500

    return jsonify({"message": "User created and welcome email sent"}), 201

if __name__ == '__main__':
    # Development server; use a proper WSGI server in production
    app.run(host='0.0.0.0', port=5000, debug=True)
```
