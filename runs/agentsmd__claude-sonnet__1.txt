```js
// lib/agentmail.js
const { AgentMailClient } = require("agentmail");

const agentmail = new AgentMailClient({
  apiKey: process.env.AGENTMAIL_API_KEY,
});

module.exports = { agentmail };
```

```js
// services/sendWelcomeEmail.js
const { agentmail } = require("../lib/agentmail");

async function sendWelcomeEmail({ to, name }) {
  await agentmail.inboxes.messages.send(process.env.AGENTMAIL_INBOX_ID, {
    to: [to],
    subject: "Welcome!",
    text: `Hi ${name || "there"},\n\nWelcome aboard — we're glad you're here.\n\n— The Team`,
    html: `<p>Hi ${name || "there"},</p><p>Welcome aboard — we're glad you're here.</p><p>— The Team</p>`,
  });
}

module.exports = { sendWelcomeEmail };
```

```js
// routes/signup.js
const express = require("express");
const { sendWelcomeEmail } = require("../services/sendWelcomeEmail");
// const { createUser } = require("../models/user"); // your existing user creation logic

const router = express.Router();

router.post("/signup", async (req, res) => {
  const { email, name, password } = req.body;

  if (!email || !password) {
    return res.status(400).json({ error: "email and password are required" });
  }

  const user = await createUser({ email, name, password });

  try {
    await sendWelcomeEmail({ to: user.email, name: user.name });
  } catch (err) {
    console.error("Failed to send welcome email:", err);
  }

  res.status(201).json({ user: { id: user.id, email: user.email, name: user.name } });
});

module.exports = router;
```
