## 1. Project layout

Create a new folder for the app (e.g. `agentmail‑example`).  
Your directory will look like this:

```
agentmail-example/
├─ src/

## 4. `src/emailService.js`

```javascript
// src/emailService.js
import { AgentMail } from 'agentmail';
import dotenv from 'dotenv';

dotenv.config(); // Load API key from .env

// Initialise the AgentMail client once
const client = new AgentMail({ apiKey: process.env.AGENTMAIL_API_KEY });

/**
 * Sends a welcome email to the newly‑registered user.
 *
 * @param {string} inboxId   - The AgentMail inbox to send from (e.g. onboarding@yourdomain.com)
 * @param {string} to        - Recipient email address
 * @param {string} name      - Recipient name (optional, used in the email body)
 */
export async function sendWelcomeEmail({ inboxId, to, name = '' }) {
  try {
    await client.inboxes.messages.send({
      inboxId,
      to: [to],
      subject: 'Welcome to Our Service!',
      // You can customise the body – plain‑text is fine for a quick welcome.
      text: `Hi ${name ? name + ',' : ''}

Thank you for signing up! We’re thrilled to have you on board.

If you have any questions, just hit reply to this email.

Best,
The Team`
    });
    console.log(`✅ Welcome email sent to ${to}`);
  } catch (err) {
    console.error('❌ Failed to send welcome email:', err);
    // Propagate the error so the route can react appropriately
    throw err;
  }
}
```

## 5. `src/app.js`

```javascript
// src/app.js
import express from 'express';
import { sendWelcomeEmail } from './emailService.js';
import dotenv from 'dotenv';

dotenv.config();

const app = express();
app.use(express.json()); // Parse JSON bodies

// Example signup payload:
// {
//   "email": "user@example.com",
//   "name": "Jane Doe"
// }
app.post('/signup', async (req, res) => {
  const { email, name } = req.body;

  if (!email) {
    return res.status(400).json({ error: 'Email is required' });
  }

  // 👉 Here you would normally create the user in your DB.
  // For this example we just simulate success.
  console.log(`🧑 New user signup: ${email}${name ? ` (${name})` : ''}`);

  try {
    await sendWelcomeEmail({
      inboxId: 'onboarding@yourdomain.com',
      to: email,
      name: name || ''
    });

    // Respond to the client after the email is queued/sent.
    res.status(201).json({ message: 'User created and welcome email sent' });
  } catch (err) {
    // If the email fails we still want the user created, but we inform the client.
    res
      .status(500)
      .json({ error: 'User created, but failed to send welcome email' });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`🚀 Express server listening on http://localhost:${PORT}`);
});
```
