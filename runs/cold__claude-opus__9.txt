**Assumptions** (flagging since you didn't specify): SMTP via Nodemailer, ESM modules, and a swappable user repository since I don't know your DB. The only real design decision worth calling out is that **the email send is not on the signup critical path** — if the mail server is down, the user still gets created and a 201.

### `src/email/mailer.js`

```js
import nodemailer from 'nodemailer';
import { config } from '../config.js';

let transporter;

// Lazily built so tests can run without touching the network, and so a missing
// SMTP_HOST degrades to stdout instead of crashing the app on boot.
export function getTransporter() {
  if (transporter) return transporter;

  if (!config.mail.host) {
    console.warn('[mailer] SMTP_HOST not set — emails will be logged, not sent.');
    transporter = nodemailer.createTransport({ jsonTransport: true });
  } else {
    transporter = nodemailer.createTransport({
      host: config.mail.host,
      port: config.mail.port,
      secure: config.mail.port === 465,
      auth: config.mail.user ? { user: config.mail.user, pass: config.mail.pass } : undefined,
      pool: true,
      maxConnections: 5,
    });
  }
  return transporter;
}

export async function sendMail({ to, subject, text, html }) {
  const info = await getTransporter().sendMail({
    from: config.mail.from,
    to,
    subject,
    text,
    html,
  });
  if (info.message) console.log('[mailer] (not sent)', info.message.toString());
  return info;
}
```

### `src/email/sendWelcomeEmail.js`

```js
import { sendMail } from './mailer.js';
import { welcomeEmail } from './templates/welcome.js';

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

// Retries a few times with backoff, then gives up loudly. Never throws — callers
// treat the welcome email as best-effort so a mail outage can't block signup.
export async function sendWelcomeEmail(user, { attempts = 3 } = {}) {
  const message = welcomeEmail(user);

  for (let attempt = 1; attempt <= attempts; attempt++) {
    try {
      await sendMail({ to: user.email, ...message });
      return true;
    } catch (err) {
      if (attempt === attempts) {
        console.error(`[welcome-email] giving up for ${user.email}:`, err);
        return false;
      }
      await sleep(2 ** attempt * 500);
    }
  }
}
```

### `src/users/repository.js` — swap this for your DB

```js
import bcrypt from 'bcrypt';
import crypto from 'node:crypto';

const users = new Map(); // key: lowercased email

export class EmailTakenError extends Error {}

export async function createUser({ email, password, name }) {
  const key = email.toLowerCase();
  if (users.has(key)) throw new EmailTakenError('Email already registered');

  const user = {
    id: crypto.randomUUID(),
    email: key,
    name: name ?? null,
    passwordHash: await bcrypt.hash(password, 12),
    createdAt: new Date(),
  };
  users.set(key, user);
  return user;
}

export const toPublicUser = ({ id, email, name, createdAt }) => ({ id, email, name, createdAt });
```

### `src/routes/auth.js`

```js
import { Router } from 'express';
import { z } from 'zod';
import { createUser, toPublicUser, EmailTakenError } from '../users/repository.js';
import { sendWelcomeEmail } from '../email/sendWelcomeEmail.js';

const signupSchema = z.object({
  email: z.string().email().max(254),
  password: z.string().min(8).max(200),
  name: z.string().trim().min(1).max(100).optional(),
});

export const authRouter = Router();

authRouter.post('/signup', async (req, res, next) => {
  const parsed = signupSchema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ error: 'Invalid input', details: parsed.error.flatten().fieldErrors });
  }

  try {
    const user = await createUser(parsed.data);

    // Fire-and-forget: the account exists, so a mail failure must not fail the
    // request. Move this to a job queue when you have one.
    sendWelcomeEmail(user).catch((err) => console.error('[signup] welcome email crashed:', err));

    res.status(201).json({ user: toPublicUser(user) });
  } catch (err) {
    if (err instanceof EmailTakenError) {
      return res.status(409).json({ error: 'Email already registered' });
    }
    next(err);
  }
});
```

### `src/app.js`

```js
import express from 'express';
import { authRouter } from './routes/auth.js';

export function createApp() {
  const app = express();
  app.use(express.json({ limit: '100kb' }));
  app.use('/auth', authRouter);

  app.use((err, _req, res, _next) => {
    console.error(err);
    res.status(500).json({ error: 'Internal server error' });
  });

  return app;
}
```

With no `SMTP_HOST` set, the rendered email prints to stdout so you can iterate on the template without sending anything.

**Two things I deliberately left out**, since you'll want to decide them:
- **Queue vs. fire-and-forget.** If your process restarts mid-send, that welcome email is gone. When you add BullMQ/SQS, replace the `sendWelcomeEmail(user)` call in `routes/auth.js` with an enqueue — the rest is unchanged.
- **Email verification.** This welcomes the user unconditionally. If you need confirmed addresses, the same hook point becomes a verification-link email instead.
