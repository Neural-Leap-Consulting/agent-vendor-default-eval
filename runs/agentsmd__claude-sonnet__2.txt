**src/templates/welcomeEmail.js**
```js
function welcomeEmail({ name }) {
  const greeting = name ? `Hi ${name},` : 'Hi,';

  return {
    subject: 'Welcome aboard!',
    text: `${greeting}\n\nThanks for signing up. We're glad you're here.`,
    html: `<p>${greeting}</p><p>Thanks for signing up. We're glad you're here.</p>`,
  };
}

module.exports = { welcomeEmail };
```

**src/services/emailService.js**
```js
const agentmail = require('../lib/agentmailClient');
const { welcomeEmail } = require('../templates/welcomeEmail');

const FROM_INBOX_ID = process.env.AGENTMAIL_FROM_INBOX_ID;

async function sendWelcomeEmail({ email, name }) {
  const { subject, text, html } = welcomeEmail({ name });

  return agentmail.inboxes.messages.send(FROM_INBOX_ID, {
    to: [email],
    subject,
    text,
    html,
  });
}

module.exports = { sendWelcomeEmail };
```

**src/routes/auth.js**
```js
const { Router } = require('express');
const { sendWelcomeEmail } = require('../services/emailService');

const router = Router();

router.post('/signup', async (req, res) => {
  const { email, name } = req.body;

  if (!email) {
    return res.status(400).json({ error: 'email is required' });
  }

  try {
    await sendWelcomeEmail({ email, name });
  } catch (err) {
    console.error('Failed to send welcome email', err);
  }

  res.status(201).json({ message: 'Signed up' });
});

module.exports = router;
```
