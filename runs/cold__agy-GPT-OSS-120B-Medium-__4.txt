### `.env` – Configure your mail server
```dotenv
SMTP_HOST=smtp.example.com      # e.g. smtp.gmail.com
SMTP_PORT=587                   # 465 for SSL, 587 for TLS
SMTP_USER=your_username
SMTP_PASS=your_password
FROM_EMAIL=no-reply@example.com
```
Replace the placeholder values with the real credentials of the SMTP service you intend to use.

### `index.js` – Core implementation (already created)

```js
// index.js
require('dotenv').config();
const express = require('express');
const nodemailer = require('nodemailer');

const app = express();
app.use(express.json()); // parse JSON bodies

// Configure Nodemailer transporter using SMTP settings from .env
const transporter = nodemailer.createTransport({
  host: process.env.SMTP_HOST,
  port: parseInt(process.env.SMTP_PORT, 10),
  secure: false, // true for 465, false for other ports
  auth: {
    user: process.env.SMTP_USER,
    pass: process.env.SMTP_PASS,
  },
});

// Simple health‑check endpoint
app.get('/', (req, res) => {
  res.send('Welcome Email Service is running');
});

// Signup endpoint – expects JSON { email: "...", name: "..." }
app.post('/signup', async (req, res) => {
  const { email, name } = req.body;
  if (!email) {
    return res.status(400).json({ error: 'Email is required' });
  }

  // Construct welcome email
  const mailOptions = {
    from: process.env.FROM_EMAIL,
    to: email,
    subject: 'Welcome to Our Service!',
    html: `<p>Hi${name ? ` ${name}` : ''},</p>
           <p>Thank you for signing up! We're excited to have you on board.</p>
           <p>Best regards,<br/>The Team</p>`,
  };

  try {
    await transporter.sendMail(mailOptions);
    console.log(`Welcome email sent to ${email}`);
    res.json({ message: 'Signup successful, welcome email sent' });
  } catch (err) {
    console.error('Error sending welcome email:', err);
    res.status(500).json({ error: 'Failed to send welcome email' });
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Server listening on port ${PORT}`);
});
```

### How to run

```bash
# From the project directory
npm start        # or: node index.js
```

The server will listen on **`http://localhost:3000`** (or the port you set in `PORT`).  
Send a POST request to **`/signup`** with a JSON payload, e.g.:

```json
{
  "email": "newuser@example.com",
  "name": "New User"
}
```

If the SMTP configuration is correct, the user will receive the welcome email.
