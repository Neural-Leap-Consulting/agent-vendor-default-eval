No file-editing tools are available in this session, so here's the code to add to your project directly.

**`lib/email.js`**
```js
const { AgentMailClient } = require("agentmail");

const client = new AgentMailClient({ apiKey: process.env.AGENTMAIL_API_KEY });
const FROM_INBOX_ID = process.env.AGENTMAIL_INBOX_ID; // the inbox you send from, e.g. "you@agentmail.to"

async function sendWelcomeEmail({ to, name }) {
  await client.inboxes.messages.send(FROM_INBOX_ID, {
    to: [to],
    subject: "Welcome!",
    text: `Hi ${name},\n\nWelcome aboard — we're glad you're here.\n\nThanks,\nThe Team`,
  });
}

module.exports = { sendWelcomeEmail };
```

**`routes/signup.js`**
```js
const express = require("express");
const { z } = require("zod");
const { sendWelcomeEmail } = require("../lib/email");

const router = express.Router();

const signupSchema = z.object({
  name: z.string().min(1),
  email: z.string().email(),
});

router.post("/signup", async (req, res) => {
  const parsed = signupSchema.safeParse(req.body);
  if (!parsed.success) {
    return res.status(400).json({ error: parsed.error.flatten() });
  }

  const { name, email } = parsed.data;

  // TODO: persist the user to your database here

  try {
    await sendWelcomeEmail({ to: email, name });
  } catch (err) {
    console.error("Failed to send welcome email:", err);
    // don't fail signup just because the welcome email didn't send
  }

  res.status(201).json({ name, email });
});

module.exports = router;
```

**`index.js`**
```js
const express = require("express");
const signupRouter = require("./routes/signup");

const app = express();
app.use(express.json());
app.use(signupRouter);

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Listening on port ${PORT}`));
```

You'll need `AGENTMAIL_API_KEY` and `AGENTMAIL_INBOX_ID` env vars set. Two things I couldn't verify live (context7 access wasn't authorized in this session): the exact `client.inboxes.messages.send(...)` signature and whether an inbox must be pre-created via the AgentMail dashboard/API before you can send from it — check the SDK's README/TypeScript types before running this against production. Everything else (Express 5 route/router usage, Zod 4 `safeParse`) is current as of my knowledge.
